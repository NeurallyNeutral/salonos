function notice(text) {
  $('#notice').textContent = text;
  clearTimeout(noticeTimer);
  noticeTimer = setTimeout(() => ($('#notice').textContent = ''), 9000);
}
async function api(path, body, method = body === undefined ? 'GET' : 'POST', headers = {}) {
  const response = await fetch(path, {
    method,
    headers: {
      ...(body === undefined ? {} : { 'Content-Type': 'application/json' }),
      ...(csrf ? { 'X-CSRF-Token': csrf } : {}),
      ...headers,
    },
    ...(body === undefined ? {} : { body: JSON.stringify(body) }),
  });
  const data = await response.json();
  if (!response.ok)
    throw Object.assign(new Error(tr(data.error)), { status: response.status, code: data.error });
  return data;
}
function action(element, fn) {
  element.addEventListener('click', async (e) => {
    e.preventDefault();
    element.disabled = true;
    try {
      await fn(e);
    } catch (err) {
      notice(err.message);
    } finally {
      element.disabled = false;
    }
  });
}
function form(element, fn) {
  element.addEventListener('submit', async (e) => {
    e.preventDefault();
    const button = $('button[type=submit]', element);
    if (button) button.disabled = true;
    try {
      await fn(new FormData(element));
    } catch (err) {
      notice(err.message);
    } finally {
      if (button) button.disabled = false;
    }
  });
}

async function agenda(target) {
  const from = Date.parse(calendarDay + 'T00:00:00Z') - 14 * 3600000,
    to = from + (calendarMode === 'week' ? 9 : 3) * 86400000;
  const data = { bookings: [] };
  let offset = 0;
  do {
    const page = await api(`/merchant/bookings?from=${from}&to=${to}&offset=${offset}`);
    data.bookings.push(...page.bookings);
    offset = page.next_offset;
  } while (offset !== null);
  const last = nextDay(calendarDay, calendarMode === 'week' ? 7 : 1),
    bookings = data.bookings.filter((b) => {
      const d = b.local_date;
      return d >= calendarDay && d < last;
    });
  target.innerHTML = `<div class="row"><div>${input('Date de l’agenda', 'calendar-date', 'date', calendarDay)}</div><button class="secondary" id="day-mode">Jour</button><button class="secondary" id="week-mode">Semaine</button><span class="muted">${esc(config.tenant.timezone)} · ${bookings.length} rendez-vous</span></div>${bookings.length ? bookings.map((b) => `<article class="card booking"><div class="row between"><div><strong>${esc(b.display_starts_at)} · ${esc(b.service_name)}</strong><p>${esc(b.name)} · ${esc(b.phone)}</p></div><span class="pill">${esc({ confirmed: 'Confirmé', completed: 'Terminé', no_show: 'Absent', cancelled: 'Annulé' }[b.status])}</span></div><small>${money(b.price_centimes)} · ${esc(b.source)}</small><div class="row">${b.status === 'confirmed' ? `<button class="secondary" data-attend="${b.id}" data-status="completed">Marquer terminé</button><button class="secondary" data-attend="${b.id}" data-status="no_show">Absent</button><button class="danger" data-cancel="${b.id}">Annuler</button><button class="secondary" data-move="${b.id}">Déplacer</button>` : ''}${b.status === 'completed' ? `<button class="secondary" data-payment="${b.id}" data-customer="${b.customer_id}" data-price="${b.price_centimes}">Paiement & fidélité</button>` : ''}</div></article>`).join('') : '<p class="empty">Aucun rendez-vous pour cette période.</p>'}`;
  $('#calendar-date').addEventListener('change', async (e) => {
    calendarDay = e.target.value;
    await agenda(target);
  });
  action($('#day-mode'), async () => {
    calendarMode = 'day';
    await agenda(target);
  });
  action($('#week-mode'), async () => {
    calendarMode = 'week';
    await agenda(target);
  });
  $$('[data-attend]', target).forEach((btn) =>
    action(btn, async () => {
      await api('/merchant/bookings/' + btn.dataset.attend + '/attendance', {
        status: btn.dataset.status,
      });
      await agenda(target);
    }),
  );
  $$('[data-cancel]', target).forEach((btn) =>
    action(btn, async () => {
      if (confirm('Annuler ce rendez-vous ?')) {
        await api('/merchant/bookings/' + btn.dataset.cancel + '/cancel', {});
        await agenda(target);
      }
    }),
  );
  $$('[data-move]', target).forEach((btn) =>
    action(btn, () =>
      merchantMove(
        target,
        bookings.find((b) => b.id === btn.dataset.move),
      ),
    ),
  );
  $$('[data-payment]', target).forEach((btn) =>
    action(btn, () =>
      paymentPanel(target, btn.dataset.payment, btn.dataset.customer, Number(btn.dataset.price)),
    ),
  );
}

async function start() {
  try {
    environment = await api('/app-config');
    if (environment.local) {
      $('#environment').hidden = false;
      $('#environment').textContent =
        'DÉMO LOCALE · Données fictives · E-mail et WhatsApp simulés · Aucun paiement traité';
      $('#demo-link').hidden = false;
    }
    const path = location.pathname;
    if (await launchStart(path, { api, session })) return;
    if (path === '/') await home();
    else if (path === '/login') await login();
    else if (path === '/inbox') await inbox();
    else if (path.startsWith('/book/')) await bookingPage(decodeURIComponent(path.slice(6)));
    else if (path === '/manage') await managePage();
    else if (path === '/dashboard') await dashboard();
    else if (path === '/manual') {
      await session();
      await bookingPage(config.tenant.slug, { manual: true });
    } else if (path === '/privacy') await privacy();
    else if (path === '/unsubscribe') {
      main.innerHTML =
        title('Vos préférences', 'Suspendre les messages') +
        '<button id="unsubscribe">Confirmer la désinscription</button>';
      action($('#unsubscribe'), async () => {
        const [customer_id, signature] = location.hash.slice(1).split('.');
        await api('/unsubscribe', { customer_id, signature });
        main.innerHTML = title('Préférences enregistrées', 'Les messages sont suspendus.');
      });
    } else
      main.innerHTML =
        title('Page introuvable', 'Retrouvons votre chemin.') + '<a href="/">Accueil</a>';
  } catch (err) {
    if (err.status === 401) {
      location.href = '/login';
      return;
    }
    main.innerHTML = `<section class="narrow">${title('Une difficulté est survenue', err.message)}<a href="/">Retour à l’accueil</a></section>`;
  }
}
await start();
