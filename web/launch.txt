const esc = (v) =>
  String(v ?? '').replace(
    /[&<>"']/g,
    (ch) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[ch],
  );
const money = (n) =>
  new Intl.NumberFormat('fr-MA', { style: 'currency', currency: 'MAD' }).format(n / 100);
const field = (label, name, type = 'text', value = '', extra = '') =>
  `<label>${esc(label)}<input name="${name}" type="${type}" value="${esc(value)}" ${extra}></label>`;
const button = (text) => `<button type="submit">${text}</button>`;
let api, session, main;
function onform(selector, fn) {
  const form = main.querySelector(selector);
  form.addEventListener('submit', async (event) => {
    event.preventDefault();
    const submit = form.querySelector('button[type=submit]');
    submit.disabled = true;
    try {
      await fn(new FormData(form));
    } catch (e) {
      document.querySelector('#notice').textContent = e.message;
    } finally {
      submit.disabled = false;
    }
  });
}
export async function launchStart(path, dependencies) {
  ({ api, session } = dependencies);
  main = document.querySelector('#main');
  if (path.startsWith('/shop/')) {
    await shop(decodeURIComponent(path.slice(6)));
    return true;
  }
  if (path === '/launch') {
    await session();
    const data = await api('/merchant/launch');
    if (data.tenant.storefront_status === 'draft') await setup();
    else await retail();
    return true;
  }
  if (path === '/setup') {
    await setup();
    return true;
  }
  if (path === '/retention') {
    await retentionPage();
    return true;
  }
  if (path === '/insights') {
    await insightsPage();
    return true;
  }
  if (path === '/signup') {
    await signupPage();
    return true;
  }
  if (path === '/preview') {
    const data = await api('/preview-api', { token: location.hash.slice(1) });
    main.innerHTML =
      branded(data, true) +
      '<p>Aperçu autorisé · Aucune réservation, demande ou vente possible.</p>' +
      data.items
        .map(
          (i) =>
            `<article class="card"><h2>${esc(i.name)}</h2><p>${esc(i.description)}</p></article>`,
        )
        .join('');
    return true;
  }
  return false;
}

function cents(v) {
  const s = String(v).replace(',', '.');
  if (!/^\d+(?:\.\d{1,2})?$/.test(s))
    throw Error('Utilisez un montant avec deux décimales au maximum.');
  const [major, minor = ''] = s.split('.');
  return Number(major) * 100 + Number(minor.padEnd(2, '0'));
}
