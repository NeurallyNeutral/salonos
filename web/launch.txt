const esc = (v) =>
  String(v ?? '').replace(
    /[&<>"']/g,
    (ch) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[ch],
  );
const money = (n) =>
  new Intl.NumberFormat('fr-MA', { style: 'currency', currency: 'MAD' }).format(n / 100);
const field = (label, name, type = 'text', value = '', extra = '') =>
  `<label>${esc(label)}<input name="${name}" type="${type}" value="${esc(value)}" ${extra}></label>`;
const button = (text) => `<button type="submit">${text}</button>`;
let api, session, main;
function onform(selector, fn) {
  const form = main.querySelector(selector);
  form.addEventListener('input', (event) => {
    event.target.removeAttribute('aria-invalid');
    event.target.closest('label')?.querySelector('.field-error')?.remove();
  });
  form.addEventListener('submit', async (event) => {
    event.preventDefault();
    const submit = form.querySelector('button[type=submit]');
    submit.disabled = true;
    try {
      await fn(new FormData(form));
    } catch (e) {
      if (form.id === 'asset-form' && e.code === 'invalid_choice')
        e.message = 'Choisissez une image PNG, JPEG ou WebP valide.';
      const name = {
        invalid_phone: form.elements.namedItem('phone') ? 'phone' : 'contact_phone',
        invalid_email: 'email',
        invalid_slug: 'slug',
        slug_taken: 'slug',
        image_too_large: 'image',
        invalid_image: 'image',
        invalid_upload: 'image',
        invalid_choice: form.id === 'asset-form' ? 'image' : null,
        asset_storage_unavailable: 'image',
        asset_quota: 'image',
        contact_consent_required: 'consent_contact',
      }[e.code];
      const field = name && form.elements.namedItem(name);
      if (field instanceof HTMLElement) {
        field.setAttribute('aria-invalid', 'true');
        const message = document.createElement('span');
        message.className = 'field-error';
        message.setAttribute('role', 'alert');
        message.textContent = e.message;
        field.insertAdjacentElement('afterend', message);
        field.focus();
      }
      document.querySelector('#notice').textContent = e.message;
    } finally {
      submit.disabled = false;
    }
  });
}
export async function launchStart(path, dependencies) {
  ({ api, session } = dependencies);
  main = document.querySelector('#main');
  if (path.startsWith('/shop/')) {
    await shop(decodeURIComponent(path.slice(6)));
    return true;
  }
  if (path === '/launch') {
    await session();
    const data = await api('/merchant/launch');
    if (data.tenant.storefront_status === 'draft') location.replace('/setup?step=identity');
    else await retail();
    return true;
  }
  if (path === '/setup') {
    await setup();
    return true;
  }
  if (path === '/retention') {
    await retentionPage();
    return true;
  }
  if (path === '/insights') {
    await insightsPage();
    return true;
  }
  if (path === '/connections') {
    await connectionsPage();
    return true;
  }
  if (path === '/account') {
    await accountPage();
    return true;
  }
  if (path === '/signup') {
    await signupPage();
    return true;
  }
  if (path === '/preview') {
    const data = await api('/preview-api', { token: location.hash.slice(1) });
    main.innerHTML =
      branded(data, true) +
      '<p>Aperçu autorisé · Aucune réservation, demande ou vente possible.</p>' +
      data.items
        .map(
          (i) =>
            `<article class="card"><h2>${esc(i.name)}</h2><p>${esc(i.description)}</p></article>`,
        )
        .join('');
    return true;
  }
  return false;
}

                : 'direct',
            },

        });
      }),
  );
}

        b.toString(16).padStart(2, '0'),
      ).join('');
    const { ticket } = await api('/merchant/assets/ticket', {
      mime: file.type,
      size: file.size,
      sha256: digest,
      owns_rights: d.has('rights'),
    });
    let text = '';
    for (const byte of bytes) text += String.fromCharCode(byte);
    const uploaded = await api('/merchant/assets/upload', {
      ticket,
      data: btoa(text),
      use_as_logo: d.get('purpose') === 'logo',
    });
    if (d.get('purpose') !== 'logo')
      await api(
        '/merchant/catalog/' + d.get('purpose') + '/image',
        { asset_id: uploaded.id },
        'PUT',
      );
    document.querySelector('#notice').textContent = 'Logo enregistré.';
  });
  onform('#share-preview', async (d) => {
    const r = await api('/merchant/launch/previews', {
      authorized_business_content: d.has('authorized'),
    });

      `<a href="${esc(r.url)}">Ouvrir l’aperçu autorisé</a>`;
  });
}

function cents(v) {
  const s = String(v).replace(',', '.');
  if (!/^\d+(?:\.\d{1,2})?$/.test(s))
    throw Error('Utilisez un montant avec deux décimales au maximum.');
  const [major, minor = ''] = s.split('.');
  return Number(major) * 100 + Number(minor.padEnd(2, '0'));
}
