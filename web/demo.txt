const scenes = sequences[mode];
const main = document.querySelector('#scene');
let step = 0;
let demoEmail = '';
function escape(value) {
  return String(value).replace(
    /[&<>"']/g,
    (ch) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[ch],
  );
}
function draw() {
  const [eyebrow, title, description, note, cardTitle, cardText] = scenes[step];
  const inbox = mode === 'merchant' && step === 2;
  main.innerHTML = `<section class="copy" key="${step}"><p class="eyebrow">${eyebrow}</p><h1>${title}</h1><p>${description}</p>${mode === 'merchant' && step === 1 ? '<form class="email-form" id="demo-email"><label for="email">Adresse e-mail pour la scène simulée</label><input id="email" type="email" required placeholder="vous@exemple.test" autocomplete="off"><button type="submit">Voir la vérification</button></form>' : ''}${inbox ? `<p class="simulation">Scène simulée pour ${escape(demoEmail || 'vous@exemple.test')} · aucun e-mail envoyé.</p>` : ''}<span class="fact">${note}</span></section><div class="stage" aria-hidden="true"><div class="art-card">${inbox ? '<div class="mail-icon"><span></span></div>' : '<svg class="sketch" viewBox="0 0 300 70"><path d="M8 56 C48 4 65 68 108 26 S158 55 201 20 S250 66 292 12"/></svg>'}<small>${inbox ? 'COURRIER SIMULÉ' : 'SALONOS · HISTOIRE GUIDÉE'}</small><strong>${cardTitle}</strong><div class="art-line"></div><p>${cardText}</p></div></div>`;
  document.querySelector('#count').textContent =
    `${String(step + 1).padStart(2, '0')} / ${String(scenes.length).padStart(2, '0')}`;
  const bar = document.querySelector('.progress-track');
  bar.setAttribute('aria-valuenow', String(step + 1));
  bar.setAttribute('aria-valuemax', String(scenes.length));
  document.querySelector('#progress').style.width = `${((step + 1) / scenes.length) * 100}%`;
  document.querySelector('#back').disabled = step === 0;
  document.querySelector('#next').textContent =
    step === scenes.length - 1 ? 'Terminer' : 'Continuer ↗';
  if (step === 1 && mode === 'merchant') {
    const input = document.querySelector('#email');
    input.value = demoEmail;
    document.querySelector('#demo-email').addEventListener('submit', (event) => {
      event.preventDefault();
      demoEmail = input.value;
      if (input.reportValidity()) go(1);
    });
  }
}
function go(delta) {
  step = Math.max(0, Math.min(scenes.length - 1, step + delta));
  draw();
  main.focus();
}
document.querySelector('#back').addEventListener('click', () => go(-1));
document.querySelector('#next').addEventListener('click', () => {
  if (mode === 'merchant' && step === 1) {
    document.querySelector('#demo-email').requestSubmit();
    return;
  }
  if (step === scenes.length - 1) {
    step = 0;
    draw();
    main.focus();
    return;
  }
  go(1);
});
document.querySelector('#restart').addEventListener('click', () => {
  step = 0;
  demoEmail = '';
  draw();
  main.focus();
});
document.addEventListener('keydown', (event) => {
  if (event.target instanceof HTMLInputElement) return;
  if (event.key === 'ArrowRight') document.querySelector('#next').click();
  if (event.key === 'ArrowLeft') document.querySelector('#back').click();
});
draw();
